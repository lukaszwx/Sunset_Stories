rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    function canManageGallery() {
      return request.auth != null
        && firestore.exists(/databases/(default)/documents/admins/$(request.auth.uid))
        && firestore.get(/databases/(default)/documents/admins/$(request.auth.uid)).data.active == true
        && firestore.get(/databases/(default)/documents/admins/$(request.auth.uid)).data.role in ['owner', 'editor'];
    }

    match /gallery/{allPaths=**} {
      // As imagens da galeria são públicas; somente administradores ativos podem enviar, substituir ou excluir arquivos.
      allow read: if true;
      allow create, update: if canManageGallery()
        && request.resource.size <= 8 * 1024 * 1024
        && request.resource.contentType.matches('image/(jpeg|png|webp|avif)');
      allow delete: if canManageGallery();
    }
  }
}
